1. SELECT idObiekt, plik, nazwa FROM `obiekty` WHERE panstwo = "Islandia"
2. SELECT obiekty.plik, obiekty.nazwa, obiekty.nazwaCechy, obiekty.wartoscCechy, obiekty.opis, rodzaje.rodzaj FROM `obiekty` INNER JOIN `rodzaje` ON rodzaje.idRodzaj = obiekty.idRodzaj WHERE obiekty.idObiekt = 46;
3. SELECT nazwa FROM `obiekty` WHERE panstwo = "Islandia" AND idRodzaj = 10;
4. SELECT rodzaje.rodzaj, COUNT(obiekty.idObiekt) AS "Liczba obiektów" FROM `rodzaje` INNER JOIN `obiekty` ON obiekty.idRodzaj = rodzaje.idRodzaj GROUP BY rodzaje.rodzaj;
